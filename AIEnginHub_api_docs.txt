--- TypeScript Axios / Fetch Client Setup (`src/services/api/drfClient.ts`) ---

```typescript
export const API_BASE_URL = import.meta.env.VITE_DRF_API_BASE || 'http://localhost:8000/api/v1';

class DRFClient {
  private accessToken: string | null = localStorage.getItem('ila_access_token');
  private refreshToken: string | null = localStorage.getItem('ila_refresh_token');

  public setTokens(access: string, refresh: string) {
    this.accessToken = access;
    this.refreshToken = refresh;
    localStorage.setItem('ila_access_token', access);
    localStorage.setItem('ila_refresh_token', refresh);
  }

  public clearTokens() {
    this.accessToken = null;
    this.refreshToken = null;
    localStorage.removeItem('ila_access_token');
    localStorage.removeItem('ila_refresh_token');
  }

  public async request<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
    const url = `${API_BASE_URL}${endpoint.startsWith('/') ? '' : '/'}${endpoint}`;
    const headers: Record<string, string> = {
      'Content-Type': 'application/json',
      ...(options.headers as Record<string, string> || {}),
    };

    if (this.accessToken) {
      headers['Authorization'] = `Bearer ${this.accessToken}`;
    }

    const res = await fetch(url, { ...options, headers });
    if (!res.ok) {
      const errorBody = await res.json().catch(() => ({}));
      throw new Error(errorBody.error || `HTTP ${res.status}: ${res.statusText}`);
    }
    return (await res.json()) as T;
  }

  public get<T>(endpoint: string) {
    return this.request<T>(endpoint, { method: 'GET' });
  }

  public post<T>(endpoint: string, data?: any) {
    return this.request<T>(endpoint, { method: 'POST', body: data ? JSON.stringify(data) : undefined });
  }

  public put<T>(endpoint: string, data?: any) {
    return this.request<T>(endpoint, { method: 'PUT', body: data ? JSON.stringify(data) : undefined });
  }

  public patch<T>(endpoint: string, data?: any) {
    return this.request<T>(endpoint, { method: 'PATCH', body: data ? JSON.stringify(data) : undefined });
  }

  public delete<T>(endpoint: string, data?: any) {
    return this.request<T>(endpoint, { method: 'DELETE', body: data ? JSON.stringify(data) : undefined });
  }
}

export const drfClient = new DRFClient();
```

9. SECTION 8: REGIONAL LOW-LATENCY TTS AUDIO STREAM PROXY
================================================================================

Synthesizes high-fidelity regional audio with server-side caching (7 days).

Endpoint:
  GET /api/v1/tts/?q={text}&tl={target_language}
  (Also accessible at: GET /api/tts/?q=...&tl=...)

Query Parameters:
  - `q`: Text string to convert to speech (URL encoded)
  - `tl`: Target ISO language code:
      `ml` -> Malayalam (Regional Kerala Native)
      `ta` -> Tamil
      `hi` -> Hindi
      `de` -> German (Deutsch)
      `fr` -> French
      `ar` -> Arabic
      `en` -> English (Default)

Response:
  HTTP 200 OK
  Content-Type: audio/mpeg
  Cache-Control: public, max-age=604800

Frontend React Audio Playback Snippet:
```typescript
export function playRegionalSpeech(text: string, langCode: string = 'en') {
  const url = `${API_BASE_URL}/tts/?tl=${encodeURIComponent(langCode)}&q=${encodeURIComponent(text)}`;
  const audio = new Audio(url);
  audio.play().catch(err => console.error('Audio playback failed:', err));
}
```

--- Hook 1: Course Creator & Library Hook (`src/hooks/useCourses.ts`) ---

```typescript
import { useState, useEffect, useCallback } from 'react';
import { drfClient } from '../services/api/drfClient';

export interface Chapter {
  id?: string;
  chapterNumber: number;
  title: string;
  summary: string;
  content: string;
  isCompleted: boolean;
  subTopics: string[];
}

export interface Course {
  id: string;
  title: string;
  subtitle?: string;
  category: string;
  totalChapters: number;
  isFavorite: boolean;
  isPermanent: boolean;
  locked: boolean;
  chapters?: Chapter[];
}

export function useCourses() {
  const [courses, setCourses] = useState<Course[]>([]);
  const [loading, setLoading] = useState<boolean>(true);

  const fetchCourses = useCallback(async () => {
    try {
      setLoading(true);
      const res = await drfClient.get<{ results?: Course[] } | Course[]>('/courses/');
      const list = Array.isArray(res) ? res : res.results || [];
      setCourses(list);
    } catch (err) {
      console.error('Failed to load courses:', err);
    } finally {
      setLoading(false);
    }
  }, []);

  const toggleChapter = async (courseId: string, chapterId: string) => {
    try {
      const res = await drfClient.post<{ isCompleted: boolean; course: Course }>(
        `/courses/${courseId}/toggle-chapter/`,
        { chapterId }
      );
      setCourses(prev => prev.map(c => (c.id === courseId ? res.course : c)));
      return res;
    } catch (err) {
      console.error('Failed to toggle chapter:', err);
      throw err;
    }
  };

  const exportDocx = (courseId: string) => {
    window.open(`${API_BASE_URL}/courses/${courseId}/export-docx/`, '_blank');
  };

  useEffect(() => {
    fetchCourses();
  }, [fetchCourses]);

  return { courses, loading, fetchCourses, toggleChapter, exportDocx };
}
```

```typescript
import { useState } from 'react';
import { drfClient } from '../services/api/drfClient';

export function useAIEngine() {
  const [generating, setGenerating] = useState(false);

  const executeAI = async (
    prompt: string,
    productType: string = 'universal_assistant',
    model: string = 'gemini-2.5-flash'
  ) => {
    setGenerating(true);
    try {
      const res = await drfClient.post<{
        content: string;
        model: string;
        response_time_ms: number;
      }>('/ai/generate/', {
        prompt,
        productType,
        model,
      });
      return res;
    } finally {
      setGenerating(false);
    }
  };

  return { executeAI, generating };
}
```
