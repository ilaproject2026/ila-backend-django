1. ARCHITECTURAL OVERVIEW & FRONTEND CLIENT SETUP
================================================================================

Base URL Configuration:
  Development: http://127.0.0.1:8000/api/v1
  Production:  https://api.ilaglobal.edu/api/v1

Frontend Environment Variable (.env):
  VITE_API_BASE_URL=http://127.0.0.1:8000/api/v1

Global Headers Required:
  Content-Type: application/json
  Authorization: Bearer <ACCESS_TOKEN>
  X-Biometric-Token: <PIN> (Optional: Used for biometric/security gate master override)

Axios Client Configuration (`src/services/api/client.ts`):
--------------------------------------------------------------------------------
import axios from 'axios';

const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || 'http://127.0.0.1:8000/api/v1',
  headers: {
    'Content-Type': 'application/json',
  },
});

apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('ila_access_token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  const biometricPin = localStorage.getItem('ila_biometric_pin');
  if (biometricPin) {
    config.headers['X-Biometric-Token'] = biometricPin;
  }
  return config;
});

apiClient.interceptors.response.use(
  (response) => response,
  async (error) => {
    const originalRequest = error.config;
    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;
      const refresh = localStorage.getItem('ila_refresh_token');
      if (refresh) {
        try {
          const res = await axios.post(
            `${apiClient.defaults.baseURL}/auth/token/refresh/`,
            { refresh }
          );
          const newAccess = res.data.access;
          localStorage.setItem('ila_access_token', newAccess);
          originalRequest.headers.Authorization = `Bearer ${newAccess}`;
          return apiClient(originalRequest);
        } catch {
          localStorage.removeItem('ila_access_token');
          localStorage.removeItem('ila_refresh_token');
          window.location.href = '/login';
        }
      }
    }
    return Promise.reject(error);
  }
);

export default apiClient;

### 11.1. Inquiries Service (`src/services/api/inquiryService.ts`)
```typescript
import apiClient from './client';
import { Inquiry } from '../../types/ilaBackend';

export const inquiryService = {
  getAll: (params?: Record<string, string>) =>
    apiClient.get<Inquiry[]>('/inquiries/', { params }),

  getById: (id: string) =>
    apiClient.get<Inquiry>(`/inquiries/${id}/`),

  create: (data: Partial<Inquiry>) =>
    apiClient.post<Inquiry>('/inquiries/', data),

  update: (id: string, data: Partial<Inquiry>) =>
    apiClient.patch<Inquiry>(`/inquiries/${id}/`, data),

  delete: (id: string) =>
    apiClient.delete(`/inquiries/${id}/`),

  addFollowUp: (data: { inquiry: string; date: string; staff_name: string; channel: string; notes: string; outcome: string }) =>
    apiClient.post('/follow-ups/', data),
};
```

export function useInquiries(categoryFilter?: string) {
  const [inquiries, setInquiries] = useState<Inquiry[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchInquiries = async () => {
    setLoading(true);
    try {
      const params = categoryFilter ? { category: categoryFilter } : undefined;
      const res = await inquiryService.getAll(params);
      setInquiries(res.data);
      setError(null);
    } catch (err: any) {
      setError(err.response?.data?.detail || 'Failed to fetch inquiries.');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchInquiries();
  }, [categoryFilter]);

  const addInquiry = async (payload: Partial<Inquiry>) => {
    const res = await inquiryService.create(payload);
    setInquiries((prev) => [res.data, ...prev]);
    return res.data;
  };

  return { inquiries, loading, error, refresh: fetchInquiries, addInquiry };
}
